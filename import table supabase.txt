-- 1. สร้างตาราง stalls (จัดการข้อมูลล็อก/พื้นที่)[cite: 3]
CREATE TABLE stalls (
    id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    created_at TIMESTAMPTZ DEFAULT NOW(),
    stall_number TEXT,
    status TEXT DEFAULT 'available',
    description TEXT,
    display_order INT4 DEFAULT 0,
    price NUMERIC DEFAULT 0
);

-- 2. สร้างตาราง equipments (จัดการข้อมูลอุปกรณ์เสริม)[cite: 2]
CREATE TABLE equipments (
    id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    created_at TIMESTAMPTZ DEFAULT NOW(),
    description TEXT,
    price NUMERIC,
    image_url TEXT,
    name TEXT
);

-- 3. สร้างตาราง bookings (จัดการข้อมูลการจอง)[cite: 1]
CREATE TABLE bookings (
    id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    created_at TIMESTAMPTZ DEFAULT NOW(),
    booking_date TEXT,
    stall_id BIGINT REFERENCES stalls(id),
    customer_name TEXT,
    phone_number TEXT,
    time_start TEXT,
    time_end TEXT,
    booking_type TEXT,
    selected_days TEXT,
    equipment_option TEXT,
    selected_equipments JSONB,
    status TEXT DEFAULT 'pending',
    expires_at TIMESTAMPTZ DEFAULT (NOW() + INTERVAL '24 hours')
);
